cmake_minimum_required(VERSION 3.24)
project(OphionLab LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
add_executable(ophion_core_tests tests/core_tests.c)
target_include_directories(ophion_core_tests PRIVATE include)
add_test(NAME core COMMAND ophion_core_tests)
if(WIN32)
  add_library(ophion_client STATIC client/ophion_client.cpp)
  target_include_directories(ophion_client PUBLIC client include)
  target_compile_definitions(ophion_client PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
  add_executable(ophion_lab_test client/lab_test.cpp)
  target_link_libraries(ophion_lab_test PRIVATE ophion_client)
  option(OPHION_BUILD_GUI "Build the standalone ImGui/DX11 diagnostics window" ON)
  if(OPHION_BUILD_GUI)
    include(FetchContent)
    FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git
      GIT_TAG dbb5eeaadffb6a3ba6a60de1290312e5802dba5a)
    FetchContent_MakeAvailable(imgui)
    add_library(ophion_imgui STATIC ${imgui_SOURCE_DIR}/imgui.cpp
      ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp
      ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_win32.cpp
      ${imgui_SOURCE_DIR}/backends/imgui_impl_dx11.cpp)
    target_include_directories(ophion_imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
    if(MINGW)
      set(OPHION_SHADER_LIBRARY d3dcompiler_47)
    else()
      set(OPHION_SHADER_LIBRARY d3dcompiler)
    endif()
    target_link_libraries(ophion_imgui PUBLIC d3d11 dxgi ${OPHION_SHADER_LIBRARY} dwmapi)
    add_executable(ophion_console WIN32 client/console.cpp)
    target_link_libraries(ophion_console PRIVATE ophion_client ophion_imgui)
  endif()
endif()
